CREATE TABLE ER_DEPARTMENT (
    DEPT_ID NUMBER PRIMARY KEY,
    DEPT_NAME VARCHAR2(50) UNIQUE NOT NULL
);

CREATE TABLE ER_JOB (
    JOB_ID NUMBER PRIMARY KEY,
    JOB_TITLE VARCHAR2(100) NOT NULL,
    DEPT_ID NUMBER NOT NULL,
    JOB_DESCRIPTION VARCHAR2(500),
    MIN_SALARY NUMBER(10,2),
    MAX_SALARY NUMBER(10,2),
    VACANCIES NUMBER DEFAULT 1 NOT NULL,
    JOB_STATUS VARCHAR2(20) DEFAULT 'OPEN',
    POSTED_DATE DATE DEFAULT SYSDATE,
    CONSTRAINT ER_JOB_DEPT_FK
        FOREIGN KEY (DEPT_ID)
        REFERENCES ER_DEPARTMENT(DEPT_ID),
    CONSTRAINT ER_JOB_SALARY_CK
        CHECK (MIN_SALARY IS NULL OR MAX_SALARY IS NULL
               OR MIN_SALARY <= MAX_SALARY),
    CONSTRAINT ER_JOB_VACANCY_CK CHECK (VACANCIES >= 0),
    CONSTRAINT ER_JOB_STATUS_CK
        CHECK (JOB_STATUS IN ('OPEN', 'CLOSED', 'ON HOLD'))
);






CREATE TABLE ER_CANDIDATE (
    CANDIDATE_ID NUMBER PRIMARY KEY,
    FULL_NAME VARCHAR2(100) NOT NULL,
    EMAIL VARCHAR2(100) UNIQUE NOT NULL,
    PHONE VARCHAR2(15),
    QUALIFICATION VARCHAR2(100),
    EXPERIENCE_YEARS NUMBER(3,1) DEFAULT 0,
    CREATED_DATE DATE DEFAULT SYSDATE,
    CONSTRAINT ER_CAND_EXP_CK CHECK (EXPERIENCE_YEARS >= 0)
);

CREATE TABLE ER_APPLICATION (
    APPLICATION_ID NUMBER PRIMARY KEY,
    CANDIDATE_ID NUMBER NOT NULL,
    JOB_ID NUMBER NOT NULL,
    APPLICATION_DATE DATE DEFAULT SYSDATE,
    APPLICATION_STATUS VARCHAR2(25) DEFAULT 'APPLIED',
    REMARKS VARCHAR2(500),
    CONSTRAINT ER_APP_CAND_FK
        FOREIGN KEY (CANDIDATE_ID)
        REFERENCES ER_CANDIDATE(CANDIDATE_ID),
    CONSTRAINT ER_APP_JOB_FK
        FOREIGN KEY (JOB_ID)
        REFERENCES ER_JOB(JOB_ID),
    CONSTRAINT ER_APP_UNQ UNIQUE (CANDIDATE_ID, JOB_ID),
    CONSTRAINT ER_APP_STATUS_CK
        CHECK (APPLICATION_STATUS IN (
            'APPLIED', 'SHORTLISTED', 'INTERVIEW',
            'SELECTED', 'REJECTED', 'HIRED'
        ))
);




CREATE TABLE ER_INTERVIEW (
    INTERVIEW_ID NUMBER PRIMARY KEY,
    APPLICATION_ID NUMBER NOT NULL,
    INTERVIEW_ROUND NUMBER DEFAULT 1,
    INTERVIEW_DATE DATE NOT NULL,
    INTERVIEW_MODE VARCHAR2(20),
    INTERVIEWER_NAME VARCHAR2(100),
    SCORE NUMBER(5,2),
    FEEDBACK VARCHAR2(500),
    INTERVIEW_STATUS VARCHAR2(20) DEFAULT 'SCHEDULED',
    CONSTRAINT ER_INT_APP_FK
        FOREIGN KEY (APPLICATION_ID)
        REFERENCES ER_APPLICATION(APPLICATION_ID),
    CONSTRAINT ER_INT_UNQ
        UNIQUE (APPLICATION_ID, INTERVIEW_ROUND),
    CONSTRAINT ER_INT_SCORE_CK
        CHECK (SCORE IS NULL OR SCORE BETWEEN 0 AND 100),
    CONSTRAINT ER_INT_MODE_CK
        CHECK (INTERVIEW_MODE IN ('ONLINE', 'OFFLINE')),
    CONSTRAINT ER_INT_STATUS_CK
        CHECK (INTERVIEW_STATUS IN (
            'SCHEDULED', 'COMPLETED', 'CANCELLED'
        ))
);

CREATE TABLE ER_EMPLOYEE (
    EMPLOYEE_ID NUMBER PRIMARY KEY,
    APPLICATION_ID NUMBER UNIQUE NOT NULL,
    CANDIDATE_ID NUMBER UNIQUE NOT NULL,
    JOINING_DATE DATE,
    DESIGNATION VARCHAR2(100),
    ANNUAL_SALARY NUMBER(12,2),
    EMPLOYMENT_STATUS VARCHAR2(20) DEFAULT 'ACTIVE',
    CONSTRAINT ER_EMP_APP_FK
        FOREIGN KEY (APPLICATION_ID)
        REFERENCES ER_APPLICATION(APPLICATION_ID),
    CONSTRAINT ER_EMP_CAND_FK
        FOREIGN KEY (CANDIDATE_ID)
        REFERENCES ER_CANDIDATE(CANDIDATE_ID),
    CONSTRAINT ER_EMP_STATUS_CK
        CHECK (EMPLOYMENT_STATUS IN ('ACTIVE', 'INACTIVE'))
);

CREATE TABLE ER_APPLICATION_AUDIT (
    AUDIT_ID NUMBER PRIMARY KEY,
    APPLICATION_ID NUMBER NOT NULL,
    OLD_STATUS VARCHAR2(25),
    NEW_STATUS VARCHAR2(25),
    CHANGE_DATE TIMESTAMP DEFAULT SYSTIMESTAMP,
    CHANGED_BY VARCHAR2(128)
);




CREATE SEQUENCE ER_AUDIT_SEQ
START WITH 1
INCREMENT BY 1;




CREATE OR REPLACE TRIGGER ER_APP_STATUS_AUDIT
AFTER UPDATE OF APPLICATION_STATUS
ON ER_APPLICATION
FOR EACH ROW
WHEN (OLD.APPLICATION_STATUS <> NEW.APPLICATION_STATUS)
BEGIN
    INSERT INTO ER_APPLICATION_AUDIT (
        AUDIT_ID,
        APPLICATION_ID,
        OLD_STATUS,
        NEW_STATUS,
        CHANGE_DATE,
        CHANGED_BY
    )
    VALUES (
        ER_AUDIT_SEQ.NEXTVAL,
        :OLD.APPLICATION_ID,
        :OLD.APPLICATION_STATUS,
        :NEW.APPLICATION_STATUS,
        SYSTIMESTAMP,
        USER
    );
END;
/

CREATE OR REPLACE PROCEDURE ER_APPLY_FOR_JOB (
    P_CANDIDATE_ID IN NUMBER,
    P_JOB_ID IN NUMBER
)
AS
BEGIN
    INSERT INTO ER_APPLICATION (
        APPLICATION_ID,
        CANDIDATE_ID,
        JOB_ID,
        APPLICATION_DATE,
        APPLICATION_STATUS
    )
    VALUES (
        ER_APPLICATION_SEQ.NEXTVAL,
        P_CANDIDATE_ID,
        P_JOB_ID,
        SYSDATE,
        'APPLIED'
    );

    DBMS_OUTPUT.PUT_LINE(
        'Application submitted successfully.'
    );

EXCEPTION
    WHEN DUP_VAL_ON_INDEX THEN
        RAISE_APPLICATION_ERROR(
            -20002,
            'This candidate has already applied for this job, or an application ID conflicts.'
        );
    WHEN OTHERS THEN
        IF SQLCODE = -2291 THEN
            RAISE_APPLICATION_ERROR(
                -20003,
                'Candidate or job does not exist.'
            );
        ELSE
            RAISE;
        END IF;
END;

CREATE OR REPLACE PROCEDURE ER_UPDATE_STATUS (
    P_APPLICATION_ID IN NUMBER,
    P_NEW_STATUS IN VARCHAR2
)
AS
BEGIN
    UPDATE ER_APPLICATION
    SET APPLICATION_STATUS = UPPER(P_NEW_STATUS)
    WHERE APPLICATION_ID = P_APPLICATION_ID;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(
            -20004,
            'Application ID not found.'
        );
    END IF;

    DBMS_OUTPUT.PUT_LINE(
        'Application status updated successfully.'
    );
END;

CREATE OR REPLACE PROCEDURE ER_APPLICATION_REPORT
AS
    CURSOR C_APPLICATIONS IS
        SELECT
            A.APPLICATION_ID,
            C.FULL_NAME,
            J.JOB_TITLE,
            D.DEPT_NAME,
            A.APPLICATION_STATUS,
            A.APPLICATION_DATE
        FROM ER_APPLICATION A
        JOIN ER_CANDIDATE C
            ON A.CANDIDATE_ID = C.CANDIDATE_ID
        JOIN ER_JOB J
            ON A.JOB_ID = J.JOB_ID
        JOIN ER_DEPARTMENT D
            ON J.DEPT_ID = D.DEPT_ID
        ORDER BY A.APPLICATION_ID;

    V_APP_ID ER_APPLICATION.APPLICATION_ID%TYPE;
    V_NAME ER_CANDIDATE.FULL_NAME%TYPE;
    V_JOB ER_JOB.JOB_TITLE%TYPE;
    V_DEPT ER_DEPARTMENT.DEPT_NAME%TYPE;
    V_STATUS ER_APPLICATION.APPLICATION_STATUS%TYPE;
    V_DATE ER_APPLICATION.APPLICATION_DATE%TYPE;

BEGIN
    OPEN C_APPLICATIONS;

    LOOP
        FETCH C_APPLICATIONS
        INTO V_APP_ID, V_NAME, V_JOB, V_DEPT, V_STATUS, V_DATE;

        EXIT WHEN C_APPLICATIONS%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            'Application ID: ' || V_APP_ID ||
            ' | Candidate: ' || V_NAME ||
            ' | Job: ' || V_JOB ||
            ' | Department: ' || V_DEPT ||
            ' | Status: ' || V_STATUS ||
            ' | Date: ' || TO_CHAR(V_DATE, 'DD-MON-YYYY')
        );
    END LOOP;

    CLOSE C_APPLICATIONS;
END;
/

CREATE OR REPLACE PROCEDURE ER_HIRE_CANDIDATE
(
    P_APPLICATION_ID IN NUMBER,
    P_DESIGNATION IN VARCHAR2,
    P_ANNUAL_SALARY IN NUMBER
)
AS
    V_CANDIDATE_ID NUMBER;
    V_STATUS ER_APPLICATION.APPLICATION_STATUS%TYPE;
    V_EMPLOYEE_COUNT NUMBER;
BEGIN
    SELECT CANDIDATE_ID, APPLICATION_STATUS
    INTO V_CANDIDATE_ID, V_STATUS
    FROM ER_APPLICATION
    WHERE APPLICATION_ID = P_APPLICATION_ID
    FOR UPDATE;

    IF V_STATUS = 'HIRED' THEN
        RAISE_APPLICATION_ERROR(
            -20020,
            'This application is already marked as hired.'
        );
    END IF;

    SELECT COUNT(*)
    INTO V_EMPLOYEE_COUNT
    FROM ER_EMPLOYEE
    WHERE APPLICATION_ID = P_APPLICATION_ID;

    IF V_EMPLOYEE_COUNT > 0 THEN
        RAISE_APPLICATION_ERROR(
            -20021,
            'An employee record already exists for this application.'
        );
    END IF;

    UPDATE ER_APPLICATION
    SET APPLICATION_STATUS = 'HIRED'
    WHERE APPLICATION_ID = P_APPLICATION_ID;

    INSERT INTO ER_EMPLOYEE
    (
        EMPLOYEE_ID,
        APPLICATION_ID,
        CANDIDATE_ID,
        JOINING_DATE,
        DESIGNATION,
        ANNUAL_SALARY,
        EMPLOYMENT_STATUS
    )
    VALUES
    (
        ER_EMPLOYEE_SEQ.NEXTVAL,
        P_APPLICATION_ID,

    D.DEPT_NAME,
    COUNT(J.JOB_ID) AS TOTAL_JOBS,
    NVL(SUM(J.VACANCIES), 0) AS TOTAL_VACANCIES
FROM ER_DEPARTMENT D
LEFT JOIN ER_JOB J
    ON D.DEPT_ID = J.DEPT_ID
    AND J.JOB_STATUS = 'OPEN'
GROUP BY D.DEPT_ID, D.DEPT_NAME
ORDER BY D.DEPT_NAME;

    C.CANDIDATE_ID,
    C.FULL_NAME,
    J.JOB_TITLE,
    D.DEPT_NAME,

FROM ER_CANDIDATE C
JOIN ER_APPLICATION A
    ON C.CANDIDATE_ID = A.CANDIDATE_ID
JOIN ER_JOB J
    ON A.JOB_ID = J.JOB_ID
JOIN ER_DEPARTMENT D
    ON J.DEPT_ID = D.DEPT_ID
ORDER BY C.FULL_NAME;
